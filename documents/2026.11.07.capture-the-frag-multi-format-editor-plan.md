770	  function list(el) {
771	    const ordered = el.tagName === 'OL';
772	    let n = parseInt(el.getAttribute('start'), 10) || 1;
773	    const lines = [];
774	    for (const child of el.children) {
775	      if (child.tagName === 'UL' || child.tagName === 'OL') {
776	        lines.push(list(child).split('\n').map((l) => (l ? '  ' + l : l)).join('\n'));
777	        continue;
778	      }
779	      const marker = ordered ? `${n++}. ` : '- ';
780	      const pad = ' '.repeat(marker.length);
781	      const nested = [];
782	      const content = [];
783	      for (const c of child.childNodes) {
784	        if (c.nodeType === 1 && (c.tagName === 'UL' || c.tagName === 'OL')) nested.push(c);
785	        else content.push(c);
786	      }
787	      const text = blocksOf(content).join('\n');
788	      lines.push(text ? marker + indentRest(text, pad) : marker.trimEnd());
789	      for (const sub of nested) {
790	        lines.push(list(sub).split('\n').map((l) => (l ? pad + l : l)).join('\n'));
791	      }
792	    }
793	    return lines.join('\n');
794	  }

796	  function block(el) {
797	    switch (el.tagName) {
798	      case 'H1': case 'H2': case 'H3': case 'H4': case 'H5': case 'H6': {
799	        const level = Number(el.tagName[1]);
800	        const text = Array.from(el.childNodes).map(inline).join('').replace(/ {2}\n/g, ' ').trim();
801	        return text ? `${'#'.repeat(level)} ${text}` : '';
802	      }
803	      case 'BLOCKQUOTE': {
804	        const inner = blocksOf(el.childNodes).join('\n\n');
805	        if (!inner) return '';
806	        return inner.split('\n').map((l) => (l ? `> ${l}` : '>')).join('\n');
807	      }
808	      case 'PRE': {
809	        const code = el.innerText.replace(/​/g, '').replace(/ /g, ' ').replace(/\n+$/, '');
810	        const fence = code.includes('```') ? '~~~' : '```';
811	        return `${fence}\n${code}\n${fence}`;
812	      }
813	      case 'UL': case 'OL': return list(el);
814	      case 'HR': return '---';
815	      default: return blocksOf(el.childNodes).join('\n\n');
816	    }
817	  }

**Format adapter interface** (each adapter is plain data plus pure functions):
```js
{
  id: 'markdown', label: 'Markdown', ext: '.md', mime: 'text/markdown',
  toolbar: [ ['undo','redo'], ['block'], ['bold','italic','strike','code','link','image'], ... ],

- **Markdown**: `serialize` = the existing `toMarkdown` code (index.html:690-822), moved and parameterized on a root element. `render` = a new in-file Markdown→HTML renderer. - **HTML**: `render` = `sanitizeHtml(source)`. `serialize` = sanitized `innerHTML` with light pretty-printing. `fileContent` wraps in a minimal `<!doctype html>` document unless the source already is one.
- **Text**: `render` = escaped text in `white-space: pre-wrap`. `serialize` = `innerText`. The rich surface uses `contenteditable="plaintext-only"`.
- **Google Doc**: Google-Docs-flavored HTML (see Open Question 6). `normalize(html)` turns the clipboard markup into semantic HTML. It unwraps `<b id="docs-internal-guid-…">`, maps `font-weight:700` → `<strong>`, `font-style:italic` → `<em>`, `text-decoration:line-through` → `<s>`, and drops `<span>`/`style` noise. After normalizing it behaves like HTML. Saved as `.html`.

**`FragEditor` public API** (everything else in the app calls only this):
```
new FragEditor(hostEl, { format, mode, placeholder })

getFormat() / setFormat(id, { convert = true })   // calls FormatConverter stub
getMode()   / setMode('preview' | 'edit' | 'live')
exec(commandId, arg?) / queryState(commandId)     // routes to rich or raw surface
focus(), clear(), isEmpty(), stats() -> { words, chars }

Internals: one host holding three surfaces, with only one visible at a time.

- `<div contenteditable class="surface-live">` for **Preview/Edit**: WYSIWYG. On `input`, a debounced `serialize()` updates the source.

## Step 5: Paste detection (request 7)
`FragEditor` handles `paste` itself and calls `detectFormat({ html: cd.getData('text/html'), text: cd.getData('text/plain') })`:
1. `html` contains `docs-internal-guid` → **Google Doc**.
2. `text` parses as markup: starts with `<!doctype`/`<html`, or more than 60% of non-blank lines start with a tag → **HTML** (pasted source code).
3. A Markdown score from `text` (`^#{1,6} `, `^[-*+] `, `^\d+\. `, fences, `[x](y)`, `**x**`, `^> `, table pipes, weighted per line) ≥ threshold → **Markdown**.
4. Rich `html` present (web page, Word) → **HTML**.
5. Otherwise → **Text**.

Behavior (see Open Question 8):
- **Editor empty**: adopt the detected format. Call `setContent(normalizedSource, id)`, which updates the format chip, toolbar, and ext, and show a toast ("Detected Markdown").
- **Editor not empty, different format**: insert the fragment converted into the *current* format through the converter (HTML→MD uses the existing serializer, and any→Text uses `textContent`). The toast offers a "Switch to X" action.

## Step 6: Conversion stub (request 8)
```js
/* FORMAT CONVERTER — extension point. Register (from,to) pairs as they're built. */
const FormatConverter = {
  _fns: new Map(),
  register(from, to, fn) { this._fns.set(`${from}>${to}`, fn); },
  canConvert(from, to) { return from === to || this._fns.has(`${from}>${to}`); },
  convert(source, from, to) {
    if (from === to) return source;
    const fn = this._fns.get(`${from}>${to}`);
    if (!fn) { console.info(`[convert] ${from}→${to} not implemented`); return null; } // TODO
    return fn(source);
  },
};
```

  },
};
